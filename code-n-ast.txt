public class FullFeaturesDemo {

    // Fields
    private int id = 1;
    private static String appName = "DemoApp";

    public static void main(String[] args) {
        // 🔹 Primitive variables
        int x = 5;
        int y = 10;
        boolean active = true;
        double pi = 3.14;
        char letter = 'A';

        // 🔹 Reference type
        String message = "Hello, Java!";

        // 🔹 Arrays
        int[] numbers = {1, 2, 3, 4, 5};
        String[] names = new String[3];
        names[0] = "Alice";
        names[1] = "Bob";
        names[2] = "Charlie";

        // 🔹 Arithmetic and boolean expressions
        int sum = x + y;
        boolean check = (x < y) && active;

        // 🔹 If / else if / else
        if (x < y) {
            System.out.println("x < y");
        } else if (x == y) {
            System.out.println("x == y");
        } else {
            System.out.println("x > y");
        }

        // 🔹 Switch statement
        switch (letter) {
            case 'A':
                System.out.println("Got A");
                break;
            case 'B':
                System.out.println("Got B");
                break;
            default:
                System.out.println("Unknown letter");
        }

        // 🔹 For loop
        for (int i = 0; i < numbers.length; i++) {
            System.out.println("Number: " + numbers[i]);
        }

        // 🔹 Enhanced for loop (foreach)
        for (String name : names) {
            System.out.println("Name: " + name);
        }

        // 🔹 While loop
        int count = 0;
        while (count < 3) {
            System.out.println("While count: " + count);
            count++;
        }

        // 🔹 Do-while loop
        int doCount = 0;
        do {
            System.out.println("Do-while count: " + doCount);
            doCount++;
        } while (doCount < 2);

        // 🔹 Ternary operator
        String result = (x > y) ? "X greater" : "Y greater";

        // 🔹 Method call and return value
        int total = add(x, y);
        System.out.println("Sum: " + total);

        // 🔹 Try-catch-finally
        try {
            int div = x / 0;
        } catch (ArithmeticException e) {
            System.out.println("Error: Division by zero");
        } finally {
            System.out.println("Finally block executed");
        }

        // 🔹 Object creation and field access
        FullFeaturesDemo demo = new FullFeaturesDemo();
        System.out.println("App Name: " + appName);
        System.out.println("ID: " + demo.id);

        // 🔹 Return statement (from main just ends)
    }

    // 🔹 Method with parameters and return
    public static int add(int a, int b) {
        return a + b;
    }

    // 🔹 Method overloading
    public static double add(double a, double b) {
        return a + b;
    }

    // 🔹 Method with control flow
    public String describe(int number) {
        if (number > 0) {
            return "Positive";
        } else if (number < 0) {
            return "Negative";
        } else {
            return "Zero";
        }
    }
}
